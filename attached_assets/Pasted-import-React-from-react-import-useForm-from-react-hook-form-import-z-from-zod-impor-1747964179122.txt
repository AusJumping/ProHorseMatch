import React from "react";
import { useForm } from "react-hook-form";
import { z } from "zod";
import { zodResolver } from "@hookform/resolvers/zod";

// Define the Zod schema for validation
const formSchema = z.object({
  breed: z.string().min(1, "Breed is required"),
  discipline: z.string().min(1, "Discipline is required"),
});

type FormData = z.infer<typeof formSchema>;

export default function RequiredFieldsForm() {
  const {
    register,
    handleSubmit,
    formState: { errors },
  } = useForm<FormData>({
    resolver: zodResolver(formSchema),
  });

  const onSubmit = (data: FormData) => {
    alert("Form submitted successfully!");
    console.log(data);
  };

  return (
    <form onSubmit={handleSubmit(onSubmit)} style={{ maxWidth: 400, margin: "2rem auto" }}>
      {/* Breed Field */}
      <label htmlFor="breed">Breed:</label>
      <select id="breed" {...register("breed")} defaultValue="">
        <option value="" disabled>
          -- Select Breed --
        </option>
        <option value="thoroughbred">Thoroughbred</option>
        <option value="warmblood">Warmblood</option>
      </select>
      {errors.breed && <p style={{ color: "red" }}>{errors.breed.message}</p>}

      {/* Discipline Field */}
      <label htmlFor="discipline">Discipline:</label>
      <select id="discipline" {...register("discipline")} defaultValue="">
        <option value="" disabled>
          -- Select Discipline --
        </option>
        <option value="jumping">Jumping</option>
        <option value="dressage">Dressage</option>
      </select>
      {errors.discipline && <p style={{ color: "red" }}>{errors.discipline.message}</p>}

      {/* Submit Button */}
      <button type="submit" style={{ marginTop: "1rem" }}>
        Submit
      </button>

      {/* General error message (if needed) */}
      {(errors.breed || errors.discipline) && (
        <p style={{ color: "red", marginTop: "0.5rem" }}>
          Please complete all required fields.
        </p>
      )}
    </form>
  );
